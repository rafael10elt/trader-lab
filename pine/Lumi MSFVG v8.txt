// Função auxiliar para cálculo de EMA ou SMA
f_ma(float src, int len, string mType) =>
    mType == "EMA" ? ta.ema(src, len) : ta.sma(src, len)

ma1     = f_ma(close, lenE1, typeE1)
ma2     = f_ma(close, lenE2, typeE2)
ma3     = f_ma(close, lenE3, typeE3)
ma4     = f_ma(close, lenE4, typeE4)
vwapVal = ta.vwap(hlc3)
atrVal  = nz(ta.atr(14))

plot(showE1 ? ma1 : na, "Média 1 (9)", color=colE1, linewidth=2)
plot(showE2 ? ma2 : na, "Média 2 (20)", color=colE2, linewidth=2)
plot(showE3 ? ma3 : na, "Média 3 (50)", color=colE3, linewidth=2, style=plot.style_linebr)
plot(showE4 ? ma4 : na, "Média 4 (200)", color=colE4, linewidth=3)
plot(showVwap ? vwapVal : na, "VWAP", color=colVwap, linewidth=2)

string atrStatus = atrVal < 14.0 ? "BAIXO" : atrVal > 40.0 ? "ALTO" : "IDEAL"
color  atrCol    = atrVal < 14.0 ? color.red : atrVal > 40.0 ? color.orange : color.green

//==============================================================================
// CONTADORES DE TEMPO DAS VELAS (M1, M5, M15) EM TEMPO REAL
//==============================================================================
f_fmtTime(int totalSec) =>
    int m = int(math.floor(totalSec / 60))
    int s = totalSec % 60
    (m < 10 ? "0" : "") + str.tostring(m) + ":" + (s < 10 ? "0" : "") + str.tostring(s)

int remM1_s  = int(math.max(0, math.floor((60000 - (timenow % 60000)) / 1000)))
int remM5_s  = int(math.max(0, math.floor((300000 - (timenow % 300000)) / 1000)))
int remM15_s = int(math.max(0, math.floor((900000 - (timenow % 900000)) / 1000)))

string clockStr = "M1: " + f_fmtTime(remM1_s) + " | M5: " + f_fmtTime(remM5_s) + " | M15: " + f_fmtTime(remM15_s)

//==============================================================================
// MATRIZ HTF SEM REPAINT (BARRA FECHADA [1])
//==============================================================================
f_calcHtfBias() =>
    e20 = ta.ema(close, 20)
    e50 = ta.ema(close, 50)
    close > e20 and e20 > e50 ? 1 : (close < e20 and e20 < e50 ? -1 : 0)

int biasM5  = nz(request.security(syminfo.tickerid, "5",  f_calcHtfBias()[1], lookahead=barmerge.lookahead_off))
int biasM15 = nz(request.security(syminfo.tickerid, "15", f_calcHtfBias()[1], lookahead=barmerge.lookahead_off))
int biasM30 = nz(request.security(syminfo.tickerid, "30", f_calcHtfBias()[1], lookahead=barmerge.lookahead_off))
int biasM1  = close > ma2 and ma2 > ma3 ? 1 : (close < ma2 and ma2 < ma3 ? -1 : 0)

bool htfLongOk  = not gateHtf or (biasM15 >= 0 and biasM30 >= 0)
bool htfShortOk = not gateHtf or (biasM15 <= 0 and biasM30 <= 0)

//==============================================================================
// LÓGICA DO OPERACIONAL DANILO
//==============================================================================
// 1. Alinhamento Completo das 4 Médias
bool stackBull = ma1 > ma2 and ma2 > ma3 and ma3 > ma4 and close > vwapVal
bool stackBear = ma1 < ma2 and ma2 < ma3 and ma3 < ma4 and close < vwapVal

// 2. Rastreador de "Início do Alinhamento"
var int barsInBullAlign = 0
var int barsInBearAlign = 0

if stackBull
    barsInBullAlign += 1
else
    barsInBullAlign := 0

if stackBear
    barsInBearAlign += 1
else
    barsInBearAlign := 0

bool earlyAlignBull = not useEarlyAlign or (barsInBullAlign > 0 and barsInBullAlign <= maxBarsAlign)
bool earlyAlignBear = not useEarlyAlign or (barsInBearAlign > 0 and barsInBearAlign <= maxBarsAlign)

// 3. O "Pulo do Gato": Distância Máxima da Média de 200
float distFrom200   = math.abs(close - ma4)
bool  tooFarFrom200 = useDist200Filter and (distFrom200 > maxDist200Atr * atrVal)

// 4. Filtro de Mercado Embolado (Chop)
float fastSpread = math.abs(ma1 - ma3)
bool isEmbolado  = useChopFilter and (fastSpread < chopSpreadAtr * atrVal)

// 5. Teste / Agressão na faixa da Média 9 e 20
bool touchedMaBull = low <= ma1 and low >= (ma2 - 2.0 * syminfo.mintick * 10)
bool touchedMaBear = high >= ma1 and high <= (ma2 + 2.0 * syminfo.mintick * 10)

// 6. Barra de Ignição / Rejeição (Fechamento expressivo)
bool ignitionBull = close > open and close >= high - (high - low) * 0.35 and close > ma1
bool ignitionBear = close < open and close <= low + (high - low) * 0.35 and close < ma1

//==============================================================================
// CONTROLE DE MESA PROPRIETÁRIA
//==============================================================================
var int   dayIdentifier = 0
var int   tradesToday   = 0
var int   consecLosses  = 0
var int   lastExitTrade = 0
var float dayStartEq    = strategy.initial_capital
var float peakEodEq     = strategy.initial_capital

int curDate = year(time, "America/New_York") * 10000 + month(time, "America/New_York") * 100 + dayofmonth(time, "America/New_York")
if curDate != dayIdentifier
    dayIdentifier := curDate
    tradesToday   := 0
    consecLosses  := 0
    dayStartEq    := strategy.equity
    peakEodEq     := math.max(peakEodEq, strategy.equity)

int closedTrades = strategy.closedtrades
if closedTrades > lastExitTrade
    for i = lastExitTrade to closedTrades - 1
        float pnl = strategy.closedtrades.profit(i)
        consecLosses := pnl < 0 ? consecLosses + 1 : 0
    lastExitTrade := closedTrades

float dayPnl      = strategy.equity - dayStartEq
float propFloor   = peakEodEq - propMaxDD
float propRoom    = strategy.equity - propFloor
float lossRemain  = maxLossDay + dayPnl

bool inW1 = useW1 and not na(time(timeframe.period, w1Session, tzSession))
bool inW2 = useW2 and not na(time(timeframe.period, w2Session, tzSession))
bool timeAllowed = not useTimeFilter or (inW1 or inW2)

int nyMinNow    = hour(time, "America/New_York") * 100 + minute(time, "America/New_York")
bool isFlatTime = useFlatNY and (nyMinNow >= flatNYHour and nyMinNow < 1800)

bool riskAllowed = (not useMaxLoss or dayPnl > -maxLossDay) and (not useMaxTrds or tradesToday < maxTrdsDay) and (not useMaxConsec or consecLosses < maxConsec) and (not usePropDD or propRoom > propBuffer)

//==============================================================================
// GATILHOS FINAIS
//==============================================================================
// Este gatilho acompanha o candidato "Rompimento · 20 candles" do relatório:
// close acima/abaixo do extremo anterior e filtro direcional da média 50.
float breakoutHigh = ta.highest(high[1], breakoutLength)
float breakoutLow  = ta.lowest(low[1], breakoutLength)
bool breakoutBull  = close > breakoutHigh and close > ma3
bool breakoutBear  = close < breakoutLow and close < ma3

bool useBreakoutTrigger = triggerMode == "Rompimento 20 candles"
bool pullbackBull = stackBull and earlyAlignBull and touchedMaBull and ignitionBull and not tooFarFrom200 and not isEmbolado and htfLongOk
bool pullbackBear = stackBear and earlyAlignBear and touchedMaBear and ignitionBear and not tooFarFrom200 and not isEmbolado and htfShortOk

// Horário, limites de risco e zeragem continuam valendo para todos os gatilhos.
bool setupBull = (useBreakoutTrigger ? breakoutBull : pullbackBull) and timeAllowed and riskAllowed and not isFlatTime
bool setupBear = (useBreakoutTrigger ? breakoutBear : pullbackBear) and timeAllowed and riskAllowed and not isFlatTime

bool enterLong  = setupBull and barstate.isconfirmed and strategy.opentrades == 0
bool enterShort = setupBear and barstate.isconfirmed and strategy.opentrades == 0

plotshape(showSignalMarkers and enterLong, title="Entrada de Compra", style=shape.labelup, location=location.belowbar, color=#089981, text="COMPRA", textcolor=color.white, size=size.tiny)
plotshape(showSignalMarkers and enterShort, title="Entrada de Venda", style=shape.labeldown, location=location.abovebar, color=#f23645, text="VENDA", textcolor=color.white, size=size.tiny)
if enterLong
    alert("COMPRA confirmada · " + syminfo.ticker + " · " + triggerMode + " · fechamento " + str.tostring(close, format.mintick), alert.freq_once_per_bar_close)
if enterShort
    alert("VENDA confirmada · " + syminfo.ticker + " · " + triggerMode + " · fechamento " + str.tostring(close, format.mintick), alert.freq_once_per_bar_close)

// Caixas Visuais
type TradeBox
    box   bSL
    box   bTP
    line  lEntry
    float sl
    float tp
    bool  isLong
    bool  done

var tradeBoxes = array.new<TradeBox>()

if (enterLong or enterShort)
    bool isL = enterLong
    float entryPx = close
    
    // Stop técnico calculado com a barra de entrada e a barra anterior.
    float rawSL = isL ? (entryPx - math.min(low[0], low[1]) + 2.0) : (math.max(high[0], high[1]) - entryPx + 2.0)
    float slPts = math.min(maxSLPts, math.max(minSLPts, rawSL))
    float tpPts = slPts * rrRatio
    
    int qty = math.min(maxContracts, math.max(1, int(math.floor(riskUsd / (slPts * ptVal)))))
    
    float slPx = isL ? entryPx - slPts : entryPx + slPts
    float tpPx = isL ? entryPx + tpPts : entryPx - tpPts
    
    int slTicks = int(math.ceil(slPts / syminfo.mintick))
    int tpTicks = int(math.ceil(tpPts / syminfo.mintick))
    
    string tradeId = (isL ? "Buy_" : "Sell_") + str.tostring(bar_index)
    
    if opMode == "Strategy (Ordens Ativas)"
        strategy.entry(tradeId, isL ? strategy.long : strategy.short, qty=qty)
        strategy.exit("Exit_" + tradeId, from_entry=tradeId, loss=slTicks, profit=tpTicks)
    
    int x1 = time
    int x2 = time + 120000
    
    string slLabel = "SL: -" + str.tostring(slPts, "#.#") + " pts (-$" + str.tostring(math.round(qty * slPts * ptVal)) + ")"
    string tpLabel = "TP: +" + str.tostring(tpPts, "#.#") + " pts (+$" + str.tostring(math.round(qty * tpPts * ptVal)) + " | " + str.tostring(rrRatio, "#.#") + "R)"
    
    box boxSL = box.new(x1, isL ? entryPx : slPx, x2, isL ? slPx : entryPx, xloc=xloc.bar_time, border_color=na, bgcolor=color.new(#f23645, 75), text=slLabel, text_size=size.tiny, text_color=color.white, text_halign=text.align_left)
    box boxTP = box.new(x1, isL ? tpPx : entryPx, x2, isL ? entryPx : tpPx, xloc=xloc.bar_time, border_color=na, bgcolor=color.new(#089981, 75), text=tpLabel, text_size=size.tiny, text_color=color.white, text_halign=text.align_left)
    line lineE = line.new(x1, entryPx, x2, entryPx, xloc=xloc.bar_time, color=color.silver, style=line.style_dashed)
    
    tradeBoxes.unshift(TradeBox.new(boxSL, boxTP, lineE, slPx, tpPx, isL, false))
    tradesToday += 1

if tradeBoxes.size() > 0
    for i = 0 to tradeBoxes.size() - 1
        TradeBox tb = tradeBoxes.get(i)
        if not tb.done
            int extTime = time + 60000
            box.set_right(tb.bSL, extTime)
            box.set_right(tb.bTP, extTime)
            line.set_x2(tb.lEntry, extTime)
            
            bool hitSL = tb.isLong ? low <= tb.sl : high >= tb.sl
            bool hitTP = tb.isLong ? high >= tb.tp : low <= tb.tp
            if hitSL or hitTP or isFlatTime
                tb.done := true
    
    while tradeBoxes.size() > 10
        TradeBox oldB = tradeBoxes.pop()
        box.delete(oldB.bSL)
        box.delete(oldB.bTP)
        line.delete(oldB.lEntry)

if isFlatTime and strategy.opentrades > 0
    strategy.close_all(comment="FLAT_NY")

//==============================================================================
// 6. PAINEL DE CONTROLE (HUD COMPLETO COM FEEDBACK DO PULO DO GATO)
//==============================================================================
var tSize = hudTxtSize == "Tiny" ? size.tiny : hudTxtSize == "Small" ? size.small : size.normal

string stTxt = triggerMode == "Rompimento 20 candles" ? "AGUARDANDO ROMPIMENTO 20" : "AGUARDANDO PULLBACK"
color  stBg  = #1e222d
color  stFg  = color.silver

if strategy.position_size > 0
    stTxt := "POSIÇÃO COMPRADA"
    stBg  := #075e4f
    stFg  := color.white
else if strategy.position_size < 0
    stTxt := "POSIÇÃO VENDIDA"
    stBg  := #7f1d1d
    stFg  := color.white
else if enterLong
    stTxt := "COMPRA CONFIRMADA (M1)"
    stBg  := #089981
    stFg  := color.white
else if enterShort
    stTxt := "VENDA CONFIRMADA (M1)"
    stBg  := #f23645
    stFg  := color.white
else if not useBreakoutTrigger and tooFarFrom200
    stTxt := "🟣 LONGE DA 200 (PULO DO GATO)"
    stBg  := #7c3aed
    stFg  := color.white
else if not useBreakoutTrigger and isEmbolado
    stTxt := "🟠 MÉDIAS EMBOLADAS (CHOP)"
    stBg  := #d97706
    stFg  := color.white
else if not useBreakoutTrigger and not earlyAlignBull and not earlyAlignBear and (stackBull or stackBear)
    stTxt := "⚪ MOVIMENTO VELHO / TARDIO"
    stBg  := #475569
    stFg  := color.silver
else if not riskAllowed
    stTxt := "⛔ TRAVA DE CONTA ACIONADA"
    stBg  := #7f1d1d
    stFg  := color.white
else if not timeAllowed or isFlatTime
    stTxt := "⚪ FORA DAS JANELAS"
    stBg  := #374151
    stFg  := color.silver

float winRate = strategy.closedtrades > 0 ? (strategy.wintrades / strategy.closedtrades * 100.0) : 0.0
float pfRatio = strategy.grossloss > 0 ? (strategy.grossprofit / strategy.grossloss) : na

if barstate.islast
    string htfStr     = "M1:" + (biasM1 == 1 ? "↑" : biasM1 == -1 ? "↓" : "-") + " | M5:" + (biasM5 == 1 ? "↑" : biasM5 == -1 ? "↓" : "-") + " | M15:" + (biasM15 == 1 ? "↑" : biasM15 == -1 ? "↓" : "-") + " | M30:" + (biasM30 == 1 ? "↑" : biasM30 == -1 ? "↓" : "-")
    string pnlStr     = (dayPnl >= 0 ? "+$" : "-$") + str.tostring(math.abs(math.round(dayPnl, 2)))
    string roomStr    = "$" + str.tostring(math.round(propRoom, 0))
    string winRateStr = str.tostring(winRate, "#.#") + "%"
    string pfStr      = na(pfRatio) ? "—" : str.tostring(pfRatio, "#.##")
    string trdStr     = str.tostring(tradesToday) + "/" + str.tostring(maxTrdsDay)
    string volStr     = str.tostring(atrVal, "#.#") + " pts (" + atrStatus + ")"

    // MODO HORIZONTAL (RODAPÉ)
    if hudLayout == "Horizontal (Rodapé)"
        var table hHud = table.new(position.bottom_center, 7, 2, bgcolor=#131722, border_color=#2a2e39, border_width=1, frame_color=#2a2e39, frame_width=1)
        
        table.cell(hHud, 0, 0, "STATUS ROBÔ", bgcolor=#1e222d, text_color=color.silver, text_size=tSize)
        table.cell(hHud, 1, 0, stTxt, bgcolor=stBg, text_color=stFg, text_size=tSize)
        table.cell(hHud, 2, 0, "MATRIZ HTF", bgcolor=#1e222d, text_color=color.silver, text_size=tSize)
